cmake_minimum_required(VERSION 3.16)
project(AutodriveLab VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(autodrive src/autodrive.cpp)
target_include_directories(autodrive PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
 target_compile_options(autodrive PRIVATE /W4)
else()
 target_compile_options(autodrive PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(autodrive_demo src/main.cpp)
target_link_libraries(autodrive_demo PRIVATE autodrive)
enable_testing()
add_executable(autodrive_tests tests/test_autodrive.cpp)
target_link_libraries(autodrive_tests PRIVATE autodrive)
add_test(NAME autodrive_tests COMMAND autodrive_tests)
