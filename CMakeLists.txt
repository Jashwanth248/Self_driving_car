cmake_minimum_required(VERSION 3.16)
project(AutoDriveLab VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Use ASan and UBSan on GCC/Clang" OFF)
add_library(autodrive STATIC src/planning.cpp src/control.cpp src/perception.cpp src/simulation.cpp src/telemetry.cpp)
target_include_directories(autodrive PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(MSVC)
    target_compile_options(autodrive PRIVATE /W4)
else()
    target_compile_options(autodrive PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
    target_compile_options(autodrive PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(autodrive PUBLIC -fsanitize=address,undefined)
endif()
add_executable(autodrive_demo src/main.cpp)
target_link_libraries(autodrive_demo PRIVATE autodrive)
add_executable(autodrive_train src/train_model.cpp)
target_link_libraries(autodrive_train PRIVATE autodrive)
add_executable(autodrive_tests tests/test_autodrive.cpp)
target_link_libraries(autodrive_tests PRIVATE autodrive)
enable_testing()
add_test(NAME autodrive_tests COMMAND autodrive_tests)
add_test(NAME autodrive_scenario COMMAND autodrive_demo --model "${CMAKE_CURRENT_SOURCE_DIR}/config/risk_model.txt" --output "${CMAKE_CURRENT_BINARY_DIR}/test_telemetry.csv")
